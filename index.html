#include <TFT_eSPI.h>

TFT_eSPI screen;

void setup() {
  pinMode(2, OUTPUT);
  digitalWrite(2, HIGH);  // Turn on the screen backlight.

  screen.init();
  screen.fillScreen(TFT_BLACK);
  screen.setTextColor(TFT_WHITE);
  screen.setTextSize(2);
  screen.setCursor(42, 112);
  screen.print("Hello, world!");
}

void loop() {}
